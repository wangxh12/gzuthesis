\chapter{数学理论测试}

\section{定义与定理}

\begin{definition}
  \label{def:nonnegative}
  若实数 $x$ 满足 $x \geq 0$，
  则称 $x$ 为非负实数。
\end{definition}

\begin{theorem}[平方非负性]
  \label{thm:square}
  对任意实数 $x \in \mathbb{R}$，
  均有
  \begin{equation}
    x^2 \geq 0.
  \end{equation}
\end{theorem}

\begin{proof}
  当 $x \geq 0$ 时，$x^2 \geq 0$；
  当 $x < 0$ 时，$-x > 0$，
  因此 $x^2=(-x)^2 \geq 0$。
  故命题成立。
\end{proof}

\section{引理与交叉引用}

\begin{lemma}
  任意两个非负实数之和仍为非负实数。
\end{lemma}

根据定义~\ref{def:nonnegative} 和
定理~\ref{thm:square}，可以进一步分析
相关数学性质。
